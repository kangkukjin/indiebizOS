"""ls105-inspection — 시험 전용 점검 자료 공급자 (105회차 시험 범위).

[sense:ls105_inspection] 은 몸의 inputs/inspection_source.json(JSON 행 목록)을 읽어
items 로 반환하는 source 다. query 가 있으면 asset_id 가 그 값인 행만 반환한다.
없는 query 는 빈 items(성공), 파일 부재·파싱 실패는 실패로 구분한다.
"""
import json
from pathlib import Path

# data/packages/installed/tools/ls105-inspection/handler.py → parents[5] = 몸 루트
_BODY_ROOT = Path(__file__).resolve().parents[5]
_SOURCE_REL = "inputs/inspection_source.json"
_KEEP = ("asset_id", "checked", "source_revision")


def _source_path() -> Path:
    """공급자 원천 경로 (시험이 교체할 수 있는 단일 지점)."""
    return _BODY_ROOT / _SOURCE_REL


def _fail(code: str, message: str, path: Path) -> dict:
    return {"success": False, "error": message, "error_code": code,
            "source": str(path)}


def _load_rows(path: Path):
    """원천 행 목록 — (rows, None) 또는 (None, 실패 dict)."""
    if not path.is_file():
        return None, _fail("SOURCE_MISSING", f"점검 원천 파일이 없습니다: {path}", path)
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, UnicodeDecodeError) as e:
        return None, _fail("SOURCE_PARSE_ERROR", f"점검 원천 JSON 파싱 실패: {e}", path)
    if not isinstance(data, list) or not all(isinstance(r, dict) for r in data):
        return None, _fail("SOURCE_PARSE_ERROR", "점검 원천은 JSON 객체 행 목록이어야 합니다", path)
    return data, None


def _ls105_inspection(tool_input: dict, context) -> dict:
    path = _source_path()
    rows, err = _load_rows(path)
    if err:
        return err
    ti = tool_input or {}
    query = ti.get("query")
    if query not in (None, ""):
        rows = [r for r in rows if r.get("asset_id") == query]
    items = [{k: r[k] for k in _KEEP if k in r} | {k: v for k, v in r.items() if k not in _KEEP}
             for r in rows]
    return {"success": True, "items": items, "count": len(items),
            "query": query if query not in (None, "") else None, "source": str(path)}


def execute(tool_input: dict, context):
    name = context.tool_name
    if name == "ls105_inspection":
        return _ls105_inspection(tool_input, context)
    return {"success": False, "error": f"알 수 없는 도구: {name}"}
