"""Pure retrieval component. No gold answers, network, or report generation."""
import collections, hashlib, json, math, os, re, sys, time
from pathlib import Path
from sklearn.feature_extraction.text import TfidfVectorizer

args=json.load(sys.stdin)
t0=time.perf_counter()
chunks=[]
for doc in args['documents']:
    # Identical units for all methods; paragraphs longer than 700 chars use 600 stride.
    for para_idx,paragraph in enumerate(re.split(r'\n\s*\n',doc['text'])):
        if not paragraph.strip():continue
        for offset in range(0,len(paragraph),600):
            part=paragraph[offset:offset+700]
            chunks.append({'doc_id':doc['id'],'chunk_id':f"{doc['id']}:{para_idx}:{offset}",'text':part})

def tokens(text):
    return re.findall(r'[a-z_]+|\d+|[가-힣]+',text.lower())
counts=[collections.Counter(tokens(c['text'])) for c in chunks]
lengths=[sum(c.values()) for c in counts]
avg=sum(lengths)/len(lengths)
df=collections.Counter(term for c in counts for term in c)
idf={t:math.log(1+(len(chunks)-n+0.5)/(n+0.5)) for t,n in df.items()}
vec=TfidfVectorizer(analyzer='char_wb',ngram_range=(2,4),lowercase=True,norm='l2',smooth_idf=True,sublinear_tf=True)
matrix=vec.fit_transform([c['text'] for c in chunks])
index_seconds=time.perf_counter()-t0
results=[]
for query in args['queries']:
    qt=time.perf_counter()
    words=tokens(query['question']);bm=[]
    for c,dl in zip(counts,lengths):
        bm.append(sum(idf.get(t,0)*c[t]*2.5/(c[t]+1.5*(0.25+0.75*dl/avg)) for t in words if c[t]))
    char=(matrix@vec.transform([query['question']]).T).toarray().ravel().tolist()
    def collapse(scores):
        best={}
        for i,score in enumerate(scores):
            if score<=0:continue
            did=chunks[i]['doc_id']
            if did not in best or score>best[did][0]:best[did]=(float(score),i)
        return sorted(best.items(),key=lambda x:(-x[1][0],x[0]))
    ranking={'bm25_word':collapse(bm),'char_tfidf':collapse(char)}
    fused=collections.defaultdict(float);chosen={}
    for method,rows in ranking.items():
        for rank,(did,(score,idx)) in enumerate(rows,1):
            fused[did]+=1/(60+rank)
            if did not in chosen or method=='char_tfidf':chosen[did]=idx
    ranking['rrf']=[(did,(score,chosen[did])) for did,score in sorted(fused.items(),key=lambda x:(-x[1],x[0]))]
    elapsed=time.perf_counter()-qt
    for method,rows in ranking.items():
        top=rows[:3]
        results.append({'question_id':query['id'],'question':query['question'],'method':method,
                        'top_docs':[did for did,_ in top],
                        'contexts':[{**chunks[idx],'score':score} for did,(score,idx) in top],
                        'answer':None,'retrieval_seconds_all_methods':elapsed})
result={'items':results,'documents':len(args['documents']),'chunks':len(chunks),
        'index_seconds':index_seconds,'settings':{'bm25':{'k1':1.5,'b':0.75},'char':{'analyzer':'char_wb','ngram_range':[2,4],'sublinear_tf':True},'rrf_k':60,'top_k':3,'unit':'paragraph/max700/stride600','document_score':'max_chunk','rrf_context':'char_chunk_when_available'},
        'document_sha256':{d['id']:hashlib.sha256(d['text'].encode()).hexdigest() for d in args['documents']}}
path=Path(os.environ['INDIEBIZ_SCRIPT_RESULT']);temp=path.with_suffix('.tmp');temp.write_text(json.dumps(result,ensure_ascii=False,allow_nan=False));os.replace(temp,path)
